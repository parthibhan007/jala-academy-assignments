JavaScript Basics

1. Comments in JavaScript

// This is a single-line comment

/*
  This is a
  multi-line comment
*/

2. Array Sorting

var fruits = ["Mango", "Apple", "Banana", "Cherry"];
fruits.sort(); // alphabetical sort

var numbers = [40, 10, 30, 20, 5];
numbers.sort(function(a, b) { return a - b; }); // numeric sort

3. for/in Loop

var person = {
  name: "Alice",
  age: 22,
  course: "Computer Science"
};

for (var key in person) {
  console.log(key + ": " + person[key]);
}

4. Object Property Access

var student = {
  name: "Priya",
  age: 20,
  grade: "A"
};

// Dot notation
console.log(student.name);

// Bracket notation
console.log(student["age"]);

5. Hoisting

// Even though greet() is called BEFORE it is defined,
// it works because function declarations are hoisted.

greet(); // works! Output: Hello from hoisted function

function greet() {
  console.log("Hello from hoisted function");
}

// Variable hoisting with var:
console.log(x); // Output: undefined (not an error)
var x = 10;
console.log(x); // Output: 10

6. Strict Mode

"use strict";

// In strict mode:
// - Variables must be declared before use
// - Duplicate parameter names not allowed
// - 'this' is undefined in standalone functions

function strictDemo() {
  "use strict";
  var message = "Strict mode is ON";
  return message;
}
