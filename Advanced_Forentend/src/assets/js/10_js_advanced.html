1. The this Keyword

function Person(name, age) {
  this.name = name;
  this.age = age;
  this.greet = function() {
    return "Hi, I am " + this.name + " and I am " + this.age + " years old.";
  };
}

var p = new Person("Alice", 22);
p.greet(); // 'this' refers to the object p

2. call() and apply()

function introduce(city, country) {
  return this.name + " is from " + city + ", " + country;
}

var person1 = { name: "Priya" };
var person2 = { name: "Arjun" };

// call() - arguments passed one by one
introduce.call(person1, "Mumbai", "India");

// apply() - arguments passed as an array
introduce.apply(person2, ["Delhi", "India"]);

3. Counter Function (Closure)

function makeCounter() {
  var count = 0; // private variable

  return {
    increment: function() { count++; },
    decrement: function() { count--; },
    getCount: function() { return count; }
  };
}

var counter = makeCounter();
counter.increment();
counter.increment();
counter.decrement();
counter.getCount(); // returns 1

4. Getter and Setter

var student = {
  _name: "",
  get name() {
    return this._name;
  },
  set name(value) {
    if (value.trim() === "") {
      console.log("Name cannot be empty!");
    } else {
      this._name = value;
    }
  }
};

student.name = "Rahul"; // uses setter
console.log(student.name); // uses getter

5. Prototype

function Animal(name) {
  this.name = name;
}

// Adding a method using prototype
Animal.prototype.speak = function() {
  return this.name + " makes a sound.";
};

// All Animal objects share this method via prototype
var dog = new Animal("Dog");
var cat = new Animal("Cat");

dog.speak(); // "Dog makes a sound."
cat.speak(); // "Cat makes a sound."
